cmake_minimum_required(VERSION 3.24)
project(mac_kinect VERSION 0.1.0 LANGUAGES C CXX)
if(NOT APPLE)
    message(FATAL_ERROR "mac-kinect currently targets native macOS")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
# Pinned SDK dependencies predate CMake 4's removal of old policy support.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(cmake/Dependencies.cmake)
add_subdirectory(src/depth)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME depth_png COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests)
set_tests_properties(depth_png PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
add_executable(test_metal_filter tests/test_metal_filter.c)
target_include_directories(test_metal_filter PRIVATE src/depth)
target_link_libraries(test_metal_filter PRIVATE native_depth_metal)
set_target_properties(test_metal_filter PROPERTIES LINKER_LANGUAGE CXX)
add_test(NAME metal_filter COMMAND test_metal_filter)
set_tests_properties(metal_filter PROPERTIES SKIP_RETURN_CODE 77)
